    <script>
        // Switch Input Methods
        function switchInputTab(tab) {
            const btnUrl = document.getElementById('tab-url');
            const btnUpload = document.getElementById('tab-upload');
            const panelUrl = document.getElementById('panel-url');
            const panelUpload = document.getElementById('panel-upload');

            if (tab === 'url') {
                btnUrl.className = "px-3 py-1.5 rounded-md bg-indigo-600 text-white transition";
                btnUpload.className = "px-3 py-1.5 rounded-md text-gray-400 hover:text-white transition";
                panelUrl.classList.remove('hidden');
                panelUpload.classList.add('hidden');
            } else {
                btnUpload.className = "px-3 py-1.5 rounded-md bg-indigo-600 text-white transition";
                btnUrl.className = "px-3 py-1.5 rounded-md text-gray-400 hover:text-white transition";
                panelUpload.classList.remove('hidden');
                panelUrl.classList.add('hidden');
            }
        }

        // Switch Aspect Ratios
        function setAspectRatio(ratio) {
            ['9:16', '16:9', '1:1', '4:5'].forEach(r => {
                const btnId = 'btn-ratio-' + r.replace(':', '-');
                document.getElementById(btnId).className = "p-3 rounded-xl border border-dark-700 bg-dark-900 hover:border-dark-600 text-left transition group";
            });

            const activeBtnId = 'btn-ratio-' + ratio.replace(':', '-');
            document.getElementById(activeBtnId).className = "p-3 rounded-xl border border-indigo-500 bg-indigo-950/40 text-left transition group";
            document.getElementById('active-ratio-badge').innerText = ratio + (ratio === '9:16' ? ' Vertical' : ratio === '16:9' ? ' Cinema' : ' Square');

            const frame = document.getElementById('preview-frame');
            if (ratio === '9:16') {
                frame.style.width = '215px';
                frame.style.height = '380px';
            } else if (ratio === '16:9') {
                frame.style.width = '380px';
                frame.style.height = '214px';
            } else if (ratio === '1:1') {
                frame.style.width = '300px';
                frame.style.height = '300px';
            } else if (ratio === '4:5') {
                frame.style.width = '260px';
                frame.style.height = '325px';
            }
        }

        // Caption Styles
        let currentCaptionStyle = 'plain';
        function setCaptionStyle(style) {
            currentCaptionStyle = style;
            ['plain', 'shadow', 'multicolor'].forEach(s => {
                document.getElementById('btn-cap-' + s).className = "p-3 rounded-xl border border-dark-700 bg-dark-900 hover:border-dark-600 text-left transition";
            });
            document.getElementById('btn-cap-' + style).className = "p-3 rounded-xl border border-indigo-500 bg-indigo-950/40 text-left transition";
            updateCaptionDisplay();
        }

        function updateCaptionDisplay() {
            const l1 = document.getElementById('caption-line-1');
            const l2 = document.getElementById('caption-line-2');
            
            if (currentCaptionStyle === 'plain') {
                l1.className = "text-sm font-extrabold uppercase tracking-wide leading-tight text-white animate-pop";
                l2.className = "text-sm font-extrabold uppercase tracking-wide leading-tight text-gray-200 mt-0.5 animate-pop";
                l1.innerText = "AI transforms";
                l2.innerText = "long videos instantly";
            } else if (currentCaptionStyle === 'shadow') {
                l1.className = "text-sm font-extrabold uppercase tracking-wide leading-tight word-shadow animate-pop";
                l2.className = "text-sm font-extrabold uppercase tracking-wide leading-tight word-shadow text-yellow-400 mt-0.5 animate-pop";
                l1.innerText = "AI transforms";
                l2.innerText = "long videos instantly";
            } else if (currentCaptionStyle === 'multicolor') {
                l1.innerHTML = '<span class="word-cyan">AI</span> <span class="word-purple">transforms</span>';
                l2.innerHTML = '<span class="word-green">long</span> <span class="word-yellow">videos</span>';
            }
        }

        // Play / Pause Simulation
        let isPlaying = false;
        function togglePlayState() {
            isPlaying = !isPlaying;
            const icon = document.getElementById('play-icon');
            if (isPlaying) {
                icon.className = "fa-solid fa-pause";
            } else {
                icon.className = "fa-solid fa-play ml-0.5";
            }
        }

        function simulateLinkFetch() {
            const urlInput = document.getElementById('social-link').value;
            if(!urlInput) {
                alert('Please paste a valid video link first.');
                return;
            }
            alert('Fetching video metadata and subtitle streams successfully!');
        }

        function startAiProcessing() {
            alert('AI Pipeline Initialized: Transcribing audio, generating summary, voiceover narrator, and rendering 2-3 word captions!');
        }
